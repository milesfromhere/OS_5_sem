#include <windows.h>

static HANDLE GetOut(void) {
    return GetStdHandle(STD_OUTPUT_HANDLE);
}

static void WriteStringW(LPCWSTR s) {
    if (!s) return;
    DWORD n = 0;
    WriteConsoleW(GetOut(), s, lstrlenW(s), &n, NULL);
}

static void WriteCharW(WCHAR c) {
    DWORD n = 0;
    WriteConsoleW(GetOut(), &c, 1, &n, NULL);
}

static void WriteLineW(LPCWSTR s) {
    WriteStringW(s);
    WriteStringW(L"\r\n");
}

// Десятичное целое без знака
static void WriteDecU32(DWORD v) {
    WCHAR buf[16]; // достаточно для 32-битного числа
    int i = 0;
    if (v == 0) {
        WriteCharW(L'0');
        return;
    }
    while (v > 0 && i < (int)(sizeof(buf)/sizeof(buf[0]))) {
        buf[i++] = (WCHAR)(L'0' + (v % 10));
        v /= 10;
    }
    while (i-- > 0) WriteCharW(buf[i]);
}

// Десятичное целое со знаком для int (например приоритет потока)
static void WriteDecI32(int v) {
    if (v < 0) { WriteCharW(L'-'); v = -v; }
    WriteDecU32((DWORD)v);
}

// Шестнадцатеричное без префикса (фиксированный размер для указателя/маски)
static void WriteHexPtr(DWORD_PTR v) {
    // Выведем без лишних leading zeros? Оставим компактный вариант
    WCHAR hexDigits[] = L"0123456789ABCDEF";
    WCHAR buf[2 * sizeof(DWORD_PTR)];
    int i = 0;
    int started = 0;
    for (int shift = (int)(sizeof(DWORD_PTR) * 8) - 4; shift >= 0; shift -= 4) {
        DWORD_PTR nibble = (v >> shift) & 0xF;
        if (nibble || started) {
            buf[i++] = hexDigits[nibble];
            started = 1;
        }
    }
    if (!started) { // v == 0
        buf[i++] = L'0';
    }
    DWORD n = 0;
    WriteConsoleW(GetOut(), buf, i, &n, NULL);
}

// Печать последнего кода ошибки с префиксом (русским)
static void PrintLastErrorW(LPCWSTR prefix) {
    DWORD err = GetLastError();
    LPWSTR msg = NULL;
    DWORD flags = FORMAT_MESSAGE_ALLOCATE_BUFFER | FORMAT_MESSAGE_FROM_SYSTEM | FORMAT_MESSAGE_IGNORE_INSERTS;
    DWORD lang = MAKELANGID(LANG_NEUTRAL, SUBLANG_DEFAULT);
    FormatMessageW(flags, NULL, err, lang, (LPWSTR)&msg, 0, NULL);
    if (prefix) WriteStringW(prefix);
    if (msg) {
        WriteStringW(msg);
        LocalFree(msg);
    } else {
        WriteStringW(L"(не удалось получить текст ошибки)\r\n");
    }
}

// Печать двоичной маски (с пробелом каждые 4 бита)
static void print_binary_mask(DWORD_PTR mask) {
    int bits = (int)(sizeof(DWORD_PTR) * 8);
    for (int i = bits - 1; i >= 0; --i) {
        DWORD_PTR bit = ((mask >> i) & 1);
        WriteCharW(bit ? L'1' : L'0');
        if (i % 4 == 0 && i != 0) WriteCharW(L' ');
    }
}

// Подсчет количества установленных битов
static int count_bits(DWORD_PTR mask) {
    int cnt = 0;
    while (mask) {
        if (mask & 1) ++cnt;
        mask >>= 1;
    }
    return cnt;
}

// Строка для класса приоритета процесса (Unicode)
static LPCWSTR proc_priority_class_to_str(DWORD cls) {
    switch (cls) {
        case IDLE_PRIORITY_CLASS:          return L"IDLE_PRIORITY_CLASS";
        case BELOW_NORMAL_PRIORITY_CLASS:  return L"BELOW_NORMAL_PRIORITY_CLASS";
        case NORMAL_PRIORITY_CLASS:        return L"NORMAL_PRIORITY_CLASS";
        case ABOVE_NORMAL_PRIORITY_CLASS:  return L"ABOVE_NORMAL_PRIORITY_CLASS";
        case HIGH_PRIORITY_CLASS:          return L"HIGH_PRIORITY_CLASS";
        case REALTIME_PRIORITY_CLASS:      return L"REALTIME_PRIORITY_CLASS";
        default:                           return L"UNKNOWN_PRIORITY_CLASS";
    }
}

//
// ======= Точка входа без CRT =======
//
void WINAPI WinMainCRTStartup(void) {
    // Получение идентификаторов
    DWORD processId = GetCurrentProcessId();
    DWORD threadId  = GetCurrentThreadId();

    HANDLE hProcess = GetCurrentProcess();
    HANDLE hThread  = GetCurrentThread();

    // Класс приоритета процесса
    DWORD procPriorityClass = GetPriorityClass(hProcess);
    if (procPriorityClass == 0) {
        WriteStringW(L"Ошибка GetPriorityClass, код ошибки: ");
        WriteDecU32(GetLastError());
        WriteLineW(L"");
    }

    // Приоритет потока
    int threadPriority = GetThreadPriority(hThread);
    if (threadPriority == THREAD_PRIORITY_ERROR_RETURN) {
        WriteStringW(L"Ошибка GetThreadPriority, код ошибки: ");
        WriteDecU32(GetLastError());
        WriteLineW(L"");
    }

    // Маски родства (affinity)
    DWORD_PTR procAffinity = 0;
    DWORD_PTR sysAffinity  = 0;
    if (!GetProcessAffinityMask(hProcess, &procAffinity, &sysAffinity)) {
        WriteStringW(L"Ошибка GetProcessAffinityMask, код ошибки: ");
        WriteDecU32(GetLastError());
        WriteLineW(L"");
    }

    int availableProcessors = count_bits(procAffinity);

    // Текущий номер процессора
    DWORD currentCpu = (DWORD)-1;
    // На поддерживаемых системах доступно всегда
    currentCpu = GetCurrentProcessorNumber();

    // Вывод
    WriteStringW(L"Идентификатор процесса (PID): ");
    WriteDecU32(processId);
    WriteLineW(L"");

    WriteStringW(L"Идентификатор потока (TID): ");
    WriteDecU32(threadId);
    WriteLineW(L"");

    if (procPriorityClass != 0) {
        WriteStringW(L"Класс приоритетов процесса: ");
        WriteStringW(proc_priority_class_to_str(procPriorityClass));
        WriteStringW(L" (0x");
        WriteHexPtr(procPriorityClass);
        WriteLineW(L")");
    } else {
        WriteLineW(L"Класс приоритетов процесса: (не удалось получить)");
    }

    if (threadPriority != THREAD_PRIORITY_ERROR_RETURN) {
        WriteStringW(L"Приоритет текущего потока: ");
        WriteDecI32(threadPriority);
        WriteLineW(L" (THREAD_PRIORITY_*)");
    } else {
        WriteLineW(L"Приоритет текущего потока: (не удалось получить)");
    }

    WriteStringW(L"Маска родства процесса (processAffinityMask): 0b");
    print_binary_mask(procAffinity);
    WriteStringW(L" (0x");
    WriteHexPtr(procAffinity);
    WriteLineW(L")");

    WriteStringW(L"Системная маска родства (systemAffinityMask): 0b");
    print_binary_mask(sysAffinity);
    WriteStringW(L" (0x");
    WriteHexPtr(sysAffinity);
    WriteLineW(L")");

    WriteStringW(L"Количество процессоров, доступных для выполнения процессу: ");
    WriteDecU32((DWORD)availableProcessors);
    WriteLineW(L"");

    if (currentCpu != (DWORD)-1) {
        WriteStringW(L"Номер процессора, выполняющего текущий поток: ");
        WriteDecU32(currentCpu);
        WriteLineW(L"");
    } else {
        WriteLineW(L"Номер процессора: (не удалось получить)");
    }

    ExitProcess(0);
}

//
// ======= Приоритет процесса =======
//
static LPCWSTR proc_priority_class_to_str(DWORD cls) {
    switch (cls) {
        case IDLE_PRIORITY_CLASS:          return L"IDLE";
        case BELOW_NORMAL_PRIORITY_CLASS:  return L"BELOW_NORMAL";
        case NORMAL_PRIORITY_CLASS:        return L"NORMAL";
        case ABOVE_NORMAL_PRIORITY_CLASS:  return L"ABOVE_NORMAL";
        case HIGH_PRIORITY_CLASS:          return L"HIGH";
        case REALTIME_PRIORITY_CLASS:      return L"REALTIME";
        default:                           return L"UNKNOWN";
    }
}

//
// ======= Точка входа =======
//
void WINAPI WinMainCRTStartup(void) {
    DWORD pid = GetCurrentProcessId();
    HANDLE hProcess = GetCurrentProcess();
    HANDLE hThread  = GetCurrentThread();

    LARGE_INTEGER freq, start, end;
    QueryPerformanceFrequency(&freq);
    QueryPerformanceCounter(&start);

    for (int i = 1; i <= 1000000; i++) {
        if (i % 1000 == 0) {
            Sleep(200);

            DWORD tid = GetCurrentThreadId();
            DWORD procClass = GetPriorityClass(hProcess);
            int threadPrio  = GetThreadPriority(hThread);
            DWORD cpu       = GetCurrentProcessorNumber();

            WriteStringW(L"Итерация: ");
            WriteDecU32(i);
            WriteLineW(L"");

            WriteStringW(L"PID: ");
            WriteDecU32(pid);
            WriteLineW(L"");

            WriteStringW(L"TID: ");
            WriteDecU32(tid);
            WriteLineW(L"");

            WriteStringW(L"Класс приоритета процесса: ");
            WriteLineW(proc_priority_class_to_str(procClass));

            WriteStringW(L"Приоритет потока: ");
            WriteDecI32(threadPrio);
            WriteLineW(L"");

            WriteStringW(L"Назначенный процессор: ");
            WriteDecU32(cpu);
            WriteLineW(L"");

            WriteLineW(L"------------------------------");
        }
    }

    QueryPerformanceCounter(&end);
    double time_sec = (double)(end.QuadPart - start.QuadPart) / (double)freq.QuadPart;

    WriteLineW(L"");
    WriteStringW(L"Время выполнения: ");
    // Выводим секунды как целую и дробную часть
    DWORD whole = (DWORD)time_sec;
    WriteDecU32(whole);
    WriteCharW(L'.');
    DWORD frac = (DWORD)((time_sec - whole) * 1000); // миллисекунды
    WriteDecU32(frac);
    WriteLineW(L" сек");

    ExitProcess(0);
}

static void PrintPriorityName(DWORD cls) {
    switch (cls) {
        case REALTIME_PRIORITY_CLASS:      WriteStringW(L"REALTIME"); break;
        case HIGH_PRIORITY_CLASS:          WriteStringW(L"HIGH"); break;
        case ABOVE_NORMAL_PRIORITY_CLASS:  WriteStringW(L"ABOVE NORMAL"); break;
        case NORMAL_PRIORITY_CLASS:        WriteStringW(L"NORMAL"); break;
        case BELOW_NORMAL_PRIORITY_CLASS:  WriteStringW(L"BELOW NORMAL"); break;
        case IDLE_PRIORITY_CLASS:          WriteStringW(L"IDLE"); break;
        default:                           WriteStringW(L"UNKNOWN"); break;
    }
}

static DWORD ToPriorityClass(int p) {
    switch (p) {
        case 0: return IDLE_PRIORITY_CLASS;
        case 1: return BELOW_NORMAL_PRIORITY_CLASS;
        case 2: return NORMAL_PRIORITY_CLASS;
        case 3: return ABOVE_NORMAL_PRIORITY_CLASS;
        case 4: return HIGH_PRIORITY_CLASS;
        case 5: return REALTIME_PRIORITY_CLASS;
        default: return NORMAL_PRIORITY_CLASS;
    }
}

int wmain(int argc, wchar_t* argv[]) {
    if (argc < 4) {
        WriteLineW(L"Lab-05b: <P1_mask> <P2_priority> <P3_priority>");
        return 0;
    }

    DWORD mask = wcstoul(argv[1], NULL, 10);
    int p2 = _wtoi(argv[2]);
    int p3 = _wtoi(argv[3]);

    DWORD prio2 = ToPriorityClass(p2);
    DWORD prio3 = ToPriorityClass(p3);

    WriteLineW(L"=== Параметры ===");
    WriteStringW(L"Mask (P1): ");
    WriteDecU32(mask);
    WriteLineW(L"");

    WriteStringW(L"P2 = ");
    PrintPriorityName(prio2);
    WriteLineW(L"");

    WriteStringW(L"P3 = ");
    PrintPriorityName(prio3);
    WriteLineW(L"");
    WriteLineW(L"");

    STARTUPINFOW si1 = { sizeof(STARTUPINFOW) };
    PROCESS_INFORMATION pi1 = {0};

    STARTUPINFOW si2 = { sizeof(STARTUPINFOW) };
    PROCESS_INFORMATION pi2 = {0};

    // Команда запуска дочерних процессов
    WCHAR childCmd1[] = L"C:\\Users\\useer-b9497e4c\\source\\repos\\OS5LAB\\x64\\Debug\\Lab-05x.exe";
    WCHAR childCmd2[] = L"C:\\Users\\useer-b9497e4c\\source\\repos\\OS5LAB\\x64\\Debug\\Lab-05x.exe";

    if (!CreateProcessW(
        NULL,
        childCmd1,
        NULL, NULL, FALSE,
        CREATE_NEW_CONSOLE | CREATE_SUSPENDED,
        NULL, NULL,
        &si1, &pi1
    )) {
        WriteStringW(L"Невозможно запустить 1-ый процесс. Код = ");
        WriteDecU32(GetLastError());
        WriteLineW(L"");
        return 1;
    }

    if (!CreateProcessW(
        NULL,
        childCmd2,
        NULL, NULL, FALSE,
        CREATE_NEW_CONSOLE | CREATE_SUSPENDED,
        NULL, NULL,
        &si2, &pi2
    )) {
        WriteStringW(L"Невозможно запустить 2-ой процесс. Код = ");
        WriteDecU32(GetLastError());
        WriteLineW(L"");
        return 1;
    }

    SetProcessAffinityMask(pi1.hProcess, mask);
    SetProcessAffinityMask(pi2.hProcess, mask);
    SetPriorityClass(pi1.hProcess, prio2);
    SetPriorityClass(pi2.hProcess, prio3);

    ResumeThread(pi1.hThread);
    ResumeThread(pi2.hThread);

    WriteLineW(L"Процесс начался.");
    WriteStringW(L"PID1 = ");
    WriteDecU32(pi1.dwProcessId);
    WriteLineW(L"");

    WriteStringW(L"PID2 = ");
    WriteDecU32(pi2.dwProcessId);
    WriteLineW(L"");
    WriteLineW(L"");

    WaitForSingleObject(pi1.hProcess, INFINITE);
    WaitForSingleObject(pi2.hProcess, INFINITE);

    CloseHandle(pi1.hProcess);
    CloseHandle(pi1.hThread);
    CloseHandle(pi2.hProcess);
    CloseHandle(pi2.hThread);

    WriteLineW(L"Все процессы завершены.");
    return 0;
}

#include <windows.h>
static HANDLE GetOut(void) {
    return GetStdHandle(STD_OUTPUT_HANDLE);
}

static void WriteStringW(LPCWSTR s) {
    DWORD n;
    WriteConsoleW(GetOut(), s, lstrlenW(s), &n, NULL);
}

static void WriteLineW(LPCWSTR s) {
    WriteStringW(s);
    WriteStringW(L"\r\n");
}

static void WriteCharW(WCHAR c) {
    DWORD n;
    WriteConsoleW(GetOut(), &c, 1, &n, NULL);
}

static void WriteDecU32(DWORD v) {
    WCHAR buf[16];
    int i = 0;
    if (v == 0) { WriteCharW(L'0'); return; }
    while (v > 0 && i < 16) {
        buf[i++] = (WCHAR)(L'0' + (v % 10));
        v /= 10;
    }
    while (i-- > 0) WriteCharW(buf[i]);
}

static void WriteDecI32(int v) {
    if (v < 0) { WriteCharW(L'-'); v = -v; }
    WriteDecU32((DWORD)v);
}

static DWORD ToProcessPriority(int p) {
    switch (p) {
        case 0: return IDLE_PRIORITY_CLASS;
        case 1: return BELOW_NORMAL_PRIORITY_CLASS;
        case 2: return NORMAL_PRIORITY_CLASS;
        case 3: return ABOVE_NORMAL_PRIORITY_CLASS;
        case 4: return HIGH_PRIORITY_CLASS;
        case 5: return REALTIME_PRIORITY_CLASS;
        default: return NORMAL_PRIORITY_CLASS;
    }
}

static int ToThreadPriority(int p) {
    switch (p) {
        case 0: return THREAD_PRIORITY_IDLE;
        case 1: return THREAD_PRIORITY_LOWEST;
        case 2: return THREAD_PRIORITY_BELOW_NORMAL;
        case 3: return THREAD_PRIORITY_NORMAL;
        case 4: return THREAD_PRIORITY_ABOVE_NORMAL;
        case 5: return THREAD_PRIORITY_HIGHEST;
        case 6: return THREAD_PRIORITY_TIME_CRITICAL;
        default: return THREAD_PRIORITY_NORMAL;
    }
}

    int id;

DWORD WINAPI WorkerThread(LPVOID arg) {
    ThreadData* data = (ThreadData*)arg;

    LARGE_INTEGER freq, start, end;
    QueryPerformanceFrequency(&freq);
    QueryPerformanceCounter(&start);

    const int TOTAL_ITERS = 1000000;

    for (int i = 1; i <= TOTAL_ITERS; i++) {
        if (i % 1000 == 0) {
            Sleep(200);

            WriteStringW(L"[Поток ");
            WriteDecI32(data->id);
            WriteStringW(L"] Итерация = ");
            WriteDecU32(i);
            WriteLineW(L"");

            WriteStringW(L"  PID:            ");
            WriteDecU32(GetCurrentProcessId());
            WriteLineW(L"");

            WriteStringW(L"  TID:            ");
            WriteDecU32(GetCurrentThreadId());
            WriteLineW(L"");

            WriteStringW(L"  Класс приоритета процесса: ");
            WriteDecU32(GetPriorityClass(GetCurrentProcess()));
            WriteLineW(L"");

            WriteStringW(L"  Приоритет потока:          ");
            WriteDecI32(GetThreadPriority(GetCurrentThread()));
            WriteLineW(L"");

            WriteStringW(L"  Номер процессора:          ");
            WriteDecU32(GetCurrentProcessorNumber());
            WriteLineW(L"");

            WriteLineW(L"--------------------------------------");
        }
    }

    QueryPerformanceCounter(&end);
    double secs = (double)(end.QuadPart - start.QuadPart) / (double)freq.QuadPart;

    WriteStringW(L"[Поток ");
    WriteDecI32(data->id);
    WriteStringW(L"] Завершён за ");
    DWORD whole = (DWORD)secs;
    WriteDecU32(whole);
    WriteCharW(L'.');
    DWORD frac = (DWORD)((secs - whole) * 1000); // миллисекунды
    WriteDecU32(frac);
    WriteLineW(L" сек");

    return 0;
}

void SetAffinity(unsigned long mask) {
    cpu_set_t set;
    CPU_ZERO(&set);

    if (mask == 0) {
        long cpus = sysconf(_SC_NPROCESSORS_ONLN);
        for (int i = 0; i < cpus; i++)

void* WorkerThread(void* arg) {
    struct ThreadData* data = (struct ThreadData*)arg;

    if (setpriority(PRIO_PROCESS, 0, data->niceValue) == -1)
        fprintf(stderr, "setpriority error: %s\n", strerror(errno));

    struct timespec start, end;
    clock_gettime(CLOCK_MONOTONIC, &start);

    const int TOTAL_ITERS = 1000000;

    for (int i = 1; i <= TOTAL_ITERS; i++) {
        if (i % 1000 == 0) {
            usleep(200000); // 200 ms

            printf("[Поток %d] Итерация = %d\n", data->id, i);

                  (end.tv_nsec - start.tv_nsec) / 1e9;

Вывод!
 
1. Поясните понятия планирование и диспетчеризация?
Планирование - это стратегический процесс определения порядка выполнения задач (процессов/потоков) процессором. Решает, какая задача будет выполняться следующей.
Диспетчеризация - это тактический процесс переключения контекста: сохранение состояния текущей задачи, загрузка состояния следующей задачи и передача ей управления процессором.
2. Опишите процесс диспетчеризации?
Процесс диспетчеризации включает:
1.	Сохранение контекста текущего процесса (регистры, состояние программы)
2.	Выбор следующего процесса из очереди готовых к выполнению
3.	Восстановление его контекста
4.	Передача управления выбранному процессу
5.	Обновление структур данных планировщика
3. Что такое планировщик? Какие бывают планировщики?
Планировщик - компонент ОС, распределяющий ресурсы процессора между задачами.
Виды планировщиков:
•	Долгосрочный (адмиссионный) - решает, какие процессы допустить к исполнению
•	Среднесрочный - управляет своппингом процессов между памятью и диском
•	Краткосрочный (CPU-планировщик) - распределяет процессорное время между готовыми процессами
4. Какой из планировщиков определяет степень мультизадачности ОС?
Краткосрочный планировщик определяет степень мультизадачности, так как он решает, как часто и какие процессы будут получать кванты времени CPU.
5. Какие виды алгоритмов планирования вы знаете?
•	Вытесняющие (preemptive) - ОС может прервать выполнение процесса
•	Невытесняющие (non-preemptive) - процесс выполняется до завершения или добровольного освобождения CPU
•	С приоритетами
•	Циклические (Round Robin)
•	Многоуровневые очереди
•	Многоуровневые очереди с обратной связью (MLFQ)
•	Самый короткий процесс первый (SJF)
•	Гарантированное планирование
6. Перечислите общие и специфичные задачи планирования
Общие задачи:
•	Справедливость распределения CPU
•	Максимальная утилизация процессора
•	Минимизация времени отклика
•	Минимизация времени ожидания
Специфичные задачи:
•	Обработка реального времени
•	Балансировка нагрузки в многопроцессорных системах
•	Учет энергопотребления (в мобильных системах)
7. Поясните понятие «циклическое планирование»
Циклическое планирование (Round Robin) - алгоритм, где каждому процессу выделяется фиксированный квант времени CPU. Если процесс не завершился за квант, он перемещается в конец очереди. Гарантирует справедливость, но может увеличивать время отклика.
8. Поясните понятие «приоритетное планирование»
Приоритетное планирование - алгоритм, где каждому процессу назначается приоритет. Процесс с наивысшим приоритетом выполняется первым. Может быть вытесняющим (процесс с более высоким приоритетом прерывает текущий) или невытесняющим.
9. Поясните понятие «приоритет процесса»
Приоритет процесса - числовое значение, определяющее важность процесса относительно других. Влияет на порядок получения CPU. Может быть статическим (назначается при создании) или динамическим (меняется во время выполнения).
10. Что такое базовый приоритет потока в Windows?
В Windows базовый приоритет потока вычисляется на основе:
•	Класса приоритета процесса
•	Относительного приоритета потока внутри процесса
Диапазон: от 0 (низший) до 31 (высший) для реального времени, от 1 до 15 для обычных процессов.
11. Поясните принцип использования значения nice в Linux
nice - значение "любезности" процесса в Linux, влияющее на его приоритет. Диапазон: от -20 (наивысший приоритет) до 19 (наинизший). Чем выше значение nice, тем "любезнее" процесс (реже получает CPU). Только суперпользователь может устанавливать отрицательные значения.
12. Поясните понятие «кооперативное планирование»
Кооперативное планирование - невытесняющий метод, где процесс добровольно освобождает CPU (через системные вызовы, операции ввода-вывода или явный вызов yield). Проблема: зависший процесс может заблокировать всю систему.
13. Поясните понятие «родственное планирование»
Родственное планирование (affinity scheduling) - привязка процесса/потока к конкретным процессорам или ядрам. Уменьшает затраты на миграцию кэшей между процессорами, но может вызвать дисбаланс нагрузки.
14. Что такое маска родственности? Как задать?
Маска родственности - битовая маска, где каждый бит соответствует доступному процессору. Устанавливается через:
•	Windows: SetProcessAffinityMask(), SetThreadAffinityMask()
•	Linux: sched_setaffinity(), pthread_setaffinity_np()
•	Команды: taskset (Linux), start /affinity (Windows)
15. Что такое перепланировка? Когда наступает момент перепланировки?
Перепланировка - активация планировщика для выбора нового процесса к выполнению.
Моменты перепланировки:
1.	Завершение процесса
2.	Блокировка процесса (ожидание I/O, семафора и т.д.)
3.	Прерывание от таймера (истечение кванта времени)
4.	Появление процесса с более высоким приоритетом
5.	Добровольное освобождение CPU (yield)
6.	Изменение приоритета или маски родственности
